@push('scripts')
<script>
    document.addEventListener('DOMContentLoaded', () => {
        const container = document.getElementById('categoryContainer');
        const leftBtn = document.getElementById('scrollLeft');
        const rightBtn = document.getElementById('scrollRight');
        const progressIndicator = document.getElementById('scrollProgress');
        const cards = container ? container.querySelectorAll('.category-card') : [];

        if (!container) return;

        if (leftBtn && rightBtn) {
            leftBtn.addEventListener('click', () => {
                const cardWidth = cards[0] ? cards[0].offsetWidth + 24 : 380;
                container.scrollBy({ left: -cardWidth, behavior: 'smooth' });
            });
            rightBtn.addEventListener('click', () => {
                const cardWidth = cards[0] ? cards[0].offsetWidth + 24 : 380;
                container.scrollBy({ left: cardWidth, behavior: 'smooth' });
            });
        }

        let isDown = false;
        let startX;
        let scrollLeftPos;
        let isDragging = false;

        container.addEventListener('mousedown', (e) => {
            isDown = true;
            isDragging = false;
            startX = e.pageX - container.offsetLeft;
            scrollLeftPos = container.scrollLeft;
        });

        container.addEventListener('mouseleave', () => isDown = false);
        container.addEventListener('mouseup', () => isDown = false);

        container.addEventListener('mousemove', (e) => {
            if (!isDown) return;
            e.preventDefault();
            const x = e.pageX - container.offsetLeft;
            const walk = (x - startX) * 1.5;
            if (Math.abs(walk) > 5) isDragging = true;
            container.scrollLeft = scrollLeftPos - walk;
        });

        cards.forEach(card => {
            card.addEventListener('click', (e) => {
                if (isDragging) e.preventDefault();
            });
        });

        container.addEventListener('keydown', (e) => {
            const cardWidth = cards[0] ? cards[0].offsetWidth + 24 : 380;
            if (e.key === 'ArrowRight') {
                e.preventDefault();
                container.scrollBy({ left: cardWidth, behavior: 'smooth' });
            } else if (e.key === 'ArrowLeft') {
                e.preventDefault();
                container.scrollBy({ left: -cardWidth, behavior: 'smooth' });
            }
        });

        const updateProgress = () => {
            if (!progressIndicator || cards.length === 0) return;
            const cardWidth = cards[0].offsetWidth + 24;
            const currentCard = Math.min(
                Math.round(container.scrollLeft / cardWidth) + 1,
                cards.length
            );
            progressIndicator.textContent = `Card ${currentCard} of ${cards.length}`;
        };

        container.addEventListener('scroll', updateProgress, { passive: true });
    });
</script>
@endpush
